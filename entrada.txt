from enum import Enum, auto


class RuntimeState(Enum):
    CREATED = auto()
    INITIALIZED = auto()
    STARTED = auto()
    STOPPING = auto()
    STOPPED = auto()


class LifecycleManager:
    """
    Controla o ciclo de vida do Runtime.
    """

    def __init__(self):

        self._state = RuntimeState.CREATED

    @property
    def state(self):

        return self._state

    def initialize(self):

        self._state = RuntimeState.INITIALIZED

    def start(self):

        self._state = RuntimeState.STARTED

    def stop(self):

        self._state = RuntimeState.STOPPED

"""
Service Registry

Responsável por registrar e resolver serviços do Runtime.

Este componente é a base da Inversão de Dependência (Dependency Inversion)
do ProjectBuilder.
"""

from __future__ import annotations

from typing import Any


class ServiceRegistry:
    """
    Registro simples de serviços.

    Mantém uma única instância para cada serviço registrado.
    """

    def __init__(self) -> None:
        self._services: dict[str, Any] = {}

    def register(self, name: str, service: Any) -> None:
        """
        Registra um serviço.

        Raises:
            ValueError:
                Caso o nome já exista.
        """

        if name in self._services:
            raise ValueError(
                f"Service '{name}' is already registered."
            )

        self._services[name] = service

    def resolve(self, name: str) -> Any:
        """
        Obtém um serviço registrado.

        Raises:
            KeyError:
                Caso o serviço não exista.
        """

        try:
            return self._services[name]
        except KeyError as exc:
            raise KeyError(
                f"Service '{name}' is not registered."
            ) from exc

    def unregister(self, name: str) -> None:
        """
        Remove um serviço.
        """

        self._services.pop(name, None)

    def exists(self, name: str) -> bool:
        """
        Verifica se o serviço existe.
        """

        return name in self._services

    def clear(self) -> None:
        """
        Remove todos os serviços.
        """

        self._services.clear()

    def list_services(self) -> list[str]:
        """
        Retorna a lista de serviços registrados.
        """

        return sorted(self._services.keys())

###############################################################################
# END FILE
###############################################################################
````

"""
Runtime Context

Mantém todos os objetos compartilhados por uma instância do Runtime.
"""

from __future__ import annotations

from dataclasses import dataclass, field
from typing import Any

from .lifecycle import LifecycleManager
from .registry import ServiceRegistry


@dataclass(slots=True)
class RuntimeContext:
    """
    Contexto principal da aplicação.

    Todos os componentes do Runtime compartilham esta estrutura.
    """

    configuration: dict[str, Any] = field(default_factory=dict)

    registry: ServiceRegistry = field(default_factory=ServiceRegistry)

    lifecycle: LifecycleManager = field(default_factory=LifecycleManager)

    metadata: dict[str, Any] = field(default_factory=dict)

    plugins: dict[str, Any] = field(default_factory=dict)

    templates: dict[str, Any] = field(default_factory=dict)

    workspace: dict[str, Any] = field(default_factory=dict)

    events: list[Any] = field(default_factory=list)

    def reset(self) -> None:
        """
        Reinicia o contexto para um estado vazio.
        """

        self.configuration.clear()
        self.registry.clear()
        self.metadata.clear()
        self.plugins.clear()
        self.templates.clear()
        self.workspace.clear()
        self.events.clear()

###############################################################################
# END FILE
###############################################################################

###############################################################################
# FILE: builder/runtime/runtime.py
###############################################################################

"""
ProjectBuilder Runtime

Este módulo implementa a fachada principal do Runtime.

Responsabilidades:

- criar o RuntimeContext
- inicializar componentes
- controlar o ciclo de vida
- fornecer acesso aos serviços registrados
"""

from __future__ import annotations

from .context import RuntimeContext
from .exceptions import RuntimeStateError
from .lifecycle import RuntimeState


class Runtime:
    """
    Fachada principal do Runtime.
    """

    def __init__(self) -> None:

        self._context = RuntimeContext()

    @property
    def context(self) -> RuntimeContext:
        return self._context

    @property
    def registry(self):
        return self._context.registry

    @property
    def lifecycle(self):
        return self._context.lifecycle

    def initialize(self) -> None:

        if self.lifecycle.state != RuntimeState.CREATED:
            raise RuntimeStateError(
                "Runtime can only be initialized once."
            )

        self.lifecycle.initialize()

    def start(self) -> None:

        if self.lifecycle.state != RuntimeState.INITIALIZED:
            raise RuntimeStateError(
                "Runtime must be initialized before starting."
            )

        self.lifecycle.start()

    def stop(self) -> None:

        if self.lifecycle.state == RuntimeState.STOPPED:
            return

        self.lifecycle.stop()

    def register(self, name: str, service: object) -> None:

        self.registry.register(name, service)

    def resolve(self, name: str):

        return self.registry.resolve(name)

###############################################################################
# END FILE
###############################################################################
````

# Refatoração do Arquivo 0004

O arquivo `builder/runtime/registry.py` deve substituir a implementação anterior por esta versão:

````python
###############################################################################
# FILE: builder/runtime/registry.py
###############################################################################

"""
Service Registry

Registro central de serviços do Runtime.
"""

from __future__ import annotations

from typing import Any, TypeVar

T = TypeVar("T")


class ServiceRegistry:
    """
    Registro simples baseado em tipos.
    """

    def __init__(self) -> None:
        self._services: dict[type[Any], Any] = {}

    def register(self, service_type: type[T], instance: T) -> None:
        if service_type in self._services:
            raise ValueError(
                f"Service '{service_type.__name__}' already registered."
            )

        self._services[service_type] = instance

    def resolve(self, service_type: type[T]) -> T:
        try:
            return self._services[service_type]
        except KeyError as exc:
            raise KeyError(
                f"Service '{service_type.__name__}' not registered."
            ) from exc

    def exists(self, service_type: type[Any]) -> bool:
        return service_type in self._services

    def unregister(self, service_type: type[Any]) -> None:
        self._services.pop(service_type, None)

    def clear(self) -> None:
        self._services.clear()

    def registered_types(self) -> tuple[type[Any], ...]:
        return tuple(sorted(self._services.keys(), key=lambda t: t.__name__))

###############################################################################
# END FILE
###############################################################################
````

from __future__ import annotations

import os

class EnvironmentProvider(ConfigurationProvider):
    """
    Carrega configurações do ambiente.
    """

    def __init__(self, prefix: str = "PROJECTBUILDER_") -> None:

        self._prefix = prefix

    def load(self) -> dict[str, str]:

        result: dict[str, str] = {}

        for key, value in os.environ.items():

            if key.startswith(self._prefix):

                name = key[len(self._prefix):].lower()

from __future__ import annotations

from collections import defaultdict
from typing import DefaultDict

from .event import Event
from .subscriber import EventSubscriber


class EventBus:
    """
    Barramento de eventos.

    Responsável apenas por registrar subscribers e publicar eventos.
    """

    def __init__(self) -> None:
        self._subscribers: DefaultDict[
            type[Event],
            list[EventSubscriber],
        ] = defaultdict(list)

    def subscribe(
        self,
        event_type: type[Event],
        subscriber: EventSubscriber,
    ) -> None:
        self._subscribers[event_type].append(subscriber)

    def subscribers_for(
        self,
        event_type: type[Event],
    ) -> tuple[EventSubscriber, ...]:
        return tuple(self._subscribers[event_type])

# Ponto de Decisão Arquitetural

Neste momento há dois caminhos possíveis:

### Opção A — Evolução incremental (mais conservadora)

Continuar com a arquitetura atual (`Runtime` como núcleo) e terminar a versão 1.0. Depois, avaliar a introdução de um `Kernel` em uma versão futura.

**Vantagens:**

- menor retrabalho;
- entrega mais rápida;
- arquitetura já parcialmente implementada.

### Opção B — Introduzir o Kernel agora (mais ambiciosa)

Reorganizar a arquitetura imediatamente para que o `Kernel` seja o ponto de entrada da plataforma.

**Vantagens:**

- base mais flexível para múltiplas engines;
- separação mais explícita de responsabilidades.

**Desvantagens:**

- exige revisar parte do que já foi modelado;
- aumenta o tempo até a primeira versão funcional.

## Minha recomendação

Eu seguiria com a **Opção A** para a versão 1.0.

A ideia do `Kernel` é interessante e pode fazer sentido no futuro, mas, neste estágio do projeto, ainda não há evidências de que a complexidade adicional trará benefícios imediatos. É melhor consolidar um **Runtime funcional**, validar a arquitetura na prática e só então decidir, com base na evolução do projeto, se a introdução de um Kernel dedicado realmente se justifica.

Essa decisão mantém o desenvolvimento incremental, reduz o risco de overengineering e preserva a possibilidade de evoluir a arquitetura no momento em que houver uma necessidade concreta.

- **Plataforma** (serviços técnicos reutilizáveis).

# Reorganização do Master Source

Depois de tudo o que discutimos, eu faria uma última melhoria.

Em vez de um único arquivo gigantesco, eu o trataria como um **artefato composto**, gerado automaticamente.

Por exemplo:

```text
ProjectBuilder_Master_Source/

00_manifest.txt

01_backlog.txt

02_rfc.txt

03_adr.txt

04_architecture.txt

05_repository.txt

06_source.txt

07_tests.txt

08_docs.txt

09_changelog.txt
```

Na etapa de geração de release, esses arquivos seriam concatenados, sempre na mesma ordem, formando:

```text
ProjectBuilder_Master_Source.txt
```

### Vantagens

- Muito mais fácil de editar.
- Menos conflitos em controle de versão.
- Geração automatizável.
- Melhor rastreabilidade.
- Mantém exatamente o mesmo resultado final.

## Minha sugestão

Em vez de manter apenas um único arquivo de texto, podemos evoluir para algo muito mais robusto:

```
ProjectBuilder/

├── ProjectBuilder_Master_Source.txt      ← gerado automaticamente
├── docs/
│   ├── architecture/
│   ├── adr/
│   ├── rfc/
│   ├── backlog/
│   ├── roadmap/
│   └── specifications/
├── builder/
├── tests/
└── tools/
    └── generate_master_source.py
```

Nesse modelo, o **`ProjectBuilder_Master_Source.txt` passa a ser gerado automaticamente** a partir do repositório, tornando-se um snapshot completo da arquitetura, documentação e código, em vez de ser editado manualmente.

Essa abordagem é mais escalável e reduz significativamente a chance de inconsistências entre documentação e implementação.
